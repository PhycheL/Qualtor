cmake_minimum_required(VERSION 3.13)
project(qualtor_gildedrose_experiment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_compile_options(-O0 -g --coverage -fprofile-abs-path -fdiagnostics-format=json -fdiagnostics-color=never)
add_link_options(--coverage)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(SOURCE_OVERRIDE)
  add_library(GildedRoseLib "${SOURCE_OVERRIDE}/GildedRose.cc")
  target_include_directories(GildedRoseLib PUBLIC "${SOURCE_OVERRIDE}")
else()
  set(BUILD_APPROVAL_TESTS_WITH_GTEST OFF CACHE BOOL "" FORCE)
  set(BUILD_UNIT_TESTS_WITH_GTEST OFF CACHE BOOL "" FORCE)
  set(BUILD_APPROVAL_TESTS_WITH_CATCH2 OFF CACHE BOOL "" FORCE)
  set(BUILD_UNIT_TESTS_WITH_CATCH2 OFF CACHE BOOL "" FORCE)
  add_subdirectory("${UPSTREAM_CPP}" upstream)
endif()
add_executable(rose_probe probe.cc)
target_link_libraries(rose_probe PRIVATE GildedRoseLib)
add_test(NAME empty_inventory COMMAND rose_probe --empty)
add_test(NAME generated_contract
  COMMAND "${Python3_EXECUTABLE}" /lab/scripts/verify.py
    --binary $<TARGET_FILE:rose_probe> --cases "${CASES_FILE}"
    --output "${RESULT_FILE}" --log-dir "${AUDIT_DIR}")
