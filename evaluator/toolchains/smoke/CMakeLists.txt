cmake_minimum_required(VERSION 3.20)
project(qualtor_cxx11_toolchain_check LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_compile_options(-O0 -g -pedantic-errors -include "${CMAKE_CURRENT_SOURCE_DIR}/cxx11_guard.h")

find_package(GTest 1.12.1 EXACT CONFIG REQUIRED)
enable_testing()
include(GoogleTest)

if(NOT UPSTREAM_CPP)
  message(FATAL_ERROR "Set UPSTREAM_CPP to the mounted upstream cpp directory")
endif()
add_library(rose STATIC "${UPSTREAM_CPP}/src/GildedRose.cc")
target_include_directories(rose PUBLIC "${UPSTREAM_CPP}/src")
add_executable(rose_tests rose_tests.cc)
target_link_libraries(rose_tests PRIVATE rose GTest::gtest_main)
gtest_discover_tests(rose_tests DISCOVERY_MODE PRE_TEST)

if(QUALTOR_MODE STREQUAL "gcc")
  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "14.2.0")
    message(FATAL_ERROR "GCC 14.2.0 is required")
  endif()
  # 只给业务代码插桩；测试框架不进入覆盖率分母。
  target_compile_options(rose PRIVATE --coverage -fprofile-abs-path)
  target_link_options(rose_tests PRIVATE --coverage)
elseif(QUALTOR_MODE STREQUAL "clang")
  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "19.1.7")
    message(FATAL_ERROR "Clang 19.1.7 is required")
  endif()
  if(NOT EXISTS "/usr/lib/mull-ir-frontend-19")
    message(FATAL_ERROR "Mull LLVM 19 plugin is missing")
  endif()
  # GoogleTest 和测试代码不加载 Mull 插件。
  target_compile_options(rose PRIVATE -fpass-plugin=/usr/lib/mull-ir-frontend-19 -grecord-command-line)
else()
  message(FATAL_ERROR "QUALTOR_MODE must be gcc or clang")
endif()
